```dart
List<Map<String, dynamic>> _getNewCases() {
  return list.where((item) {
    final status = item['status']?.toString().toLowerCase();
    final hasResponse = item['has_response'] == true;
    return status == 'new' || status == 'جديد' || !hasResponse;
  }).toList();
}
```
